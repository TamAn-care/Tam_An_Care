   180	  private stableSerialize(
   181	    value: unknown,
   182	  ): string {
   183	    const normalize = (
   184	      input: unknown,
   185	    ): unknown => {
   186	      if (Array.isArray(input)) {
   187	        return input.map(normalize);
   188	      }
   189	
   190	      if (
   191	        input !== null &&
   192	        typeof input === 'object'
   193	      ) {
   194	        return Object.fromEntries(
   195	          Object.entries(
   196	            input as Record<
   197	              string,
   198	              unknown
   199	            >,
   200	          )
   201	            .sort(
   202	              ([left], [right]) =>
   203	                left.localeCompare(
   204	                  right,
   205	                ),
   206	            )
   207	            .map(
   208	              ([key, child]) => [
   209	                key,
   210	                normalize(child),
   211	              ],
   212	            ),
   213	        );
   214	      }
   215	
   216	      return input;
   217	    };
   218	
   219	    return JSON.stringify(
   220	      normalize(value),
   221	    );
   222	  }

   224	  private async authorize(
   225	    actor: HealthReportActor,
   226	    allowed: HealthReportActorRole[],
   227	  ): Promise<void> {
   228	    if (!actor.actorId || !actor.actorRole) {
   229	      throw new UnauthorizedException(
   230	        'Human actor is required',
   231	      );
   232	    }
   233	
   234	    const result =
   235	      await this.db.query<ActorRow>(
   236	        `
   237	          SELECT
   238	            actor_id,
   239	            primary_operational_role,
   240	            status
   241	          FROM staff_actors
   242	          WHERE actor_id=$1
   243	          LIMIT 1
   244	        `,
   245	        [actor.actorId],
   246	      );
   247	
   248	    const canonical = result.rows[0];
   249	
   250	    if (
   251	      !canonical ||
   252	      canonical.status !== 'ACTIVE'
   253	    ) {
   254	      throw new UnauthorizedException(
   255	        'Canonical active human actor is required',
   256	      );
   257	    }
   258	
   259	    if (
   260	      canonical.primary_operational_role !==
   261	      actor.actorRole
   262	    ) {
   263	      throw new ForbiddenException(
   264	        'Human actor role does not match canonical role',
   265	      );
   266	    }
   267	
   268	    if (!allowed.includes(actor.actorRole)) {
   269	      throw new ForbiddenException(
   270	        'Actor is not authorized for this health report action',
   271	      );
   272	    }

   515	  private async captureSnapshotData(
   516	    residentId: string,
   517	    periodStart: Date,
   518	    periodEnd: Date,
   519	    cutoff: Date,
   520	  ): Promise<{
   521	    snapshotData:
   522	      Record<string, unknown>;
   523	    sourceManifest:
   524	      Array<Record<string, unknown>>;
   525	  }> {
   526	    const snapshotData:
   527	      Record<string, unknown> = {};
   528	
   529	    const sourceManifest:
   530	      Array<Record<string, unknown>> = [];
   531	
   532	    for (const source of this.sources) {
   533	      if (
   534	        !/^[a-z_][a-z0-9_]*$/.test(
   535	          source.table,
   536	        )
   537	      ) {
   538	        throw new BadRequestException(
   539	          'Invalid canonical source table',
   540	        );
   541	      }
   542	
   543	      let rows;
   544	
   545	      if (
   546	        source.key ===
   547	        'medicationOrders'
   548	      ) {
   549	        rows =
   550	          await this.db.query<{
   551	            row_data: unknown;
   552	          }>(
   553	            `
   554	              SELECT
   555	                to_jsonb(t) AS row_data
   556	              FROM public."${source.table}" t
   557	              WHERE
   558	                t.resident_id=$1
   559	
   560	                AND
   561	                COALESCE(
   562	                  t.effective_from,
   563	                  t.prescribed_at
   564	                ) <= $3
   565	
   566	                AND
   567	                COALESCE(
   568	                  t.effective_to,
   569	                  $3
   570	                ) >= $2
   571	
   572	                AND
   573	                t.created_at <= $4
   574	
   575	              ORDER BY
   576	                COALESCE(
   577	                  t.effective_from,
   578	                  t.prescribed_at
   579	                ) ASC,
   580	                to_jsonb(t)::text ASC
   581	            `,
   582	            [
   583	              residentId,
   584	              periodStart,
   585	              periodEnd,
   586	              cutoff,
   587	            ],
   588	          );
   589	      } else {
   590	        rows =
   591	          await this.db.query<{
   592	            row_data: unknown;
   593	          }>(
   594	            `
   595	              SELECT
   596	                to_jsonb(t) AS row_data
   597	              FROM public."${source.table}" t
   598	              WHERE
   599	                t.resident_id=$1
   600	
   601	                AND
   602	                ${source.timeExpression}
   603	                  >= $2
   604	
   605	                AND
   606	                ${source.timeExpression}
   607	                  <= $3
   608	
   609	                AND
   610	                ${source.timeExpression}
   611	                  <= $4
   612	
   613	                AND (
   614	                  NOT (
   615	                    to_jsonb(t)
   616	                    ? 'created_at'
   617	                  )
   618	                  OR (
   619	                    to_jsonb(t)
   620	                      ->> 'created_at'
   621	                  )::timestamptz
   622	                    <= $4
   623	                )
   624	
   625	              ORDER BY
   626	                ${source.timeExpression}
   627	                  ASC,
   628	                to_jsonb(t)::text ASC
   629	            `,
   630	            [
   631	              residentId,
   632	              periodStart,
   633	              periodEnd,
   634	              cutoff,
   635	            ],
   636	          );
   637	      }
   638	
   639	      snapshotData[source.key] =
   640	        rows.rows.map(
   641	          (row) => row.row_data,
   642	        );
   643	
   644	      sourceManifest.push({
   645	        sourcePolicy:
   646	          this.sourcePolicy,
   647	        key: source.key,
   648	        table: source.table,
   649	        residentKey:
   650	          'resident_id',
   651	        periodExpression:
   652	          source.key ===
   653	          'medicationOrders'
   654	            ? 'effective interval overlap'
   655	            : source.timeExpression,
   656	        cutoffPolicy:
   657	          'event-time + created_at <= source_cutoff_at',
   658	        deterministicOrder:
   659	          'event-time + canonical row JSON',
   660	        rowCount:
   661	          rows.rowCount ?? 0,
   662	      });
   663	    }
   664	
   665	    return {
   666	      snapshotData,
   667	      sourceManifest,
   668	    };
   669	  }

   671	  async generate(
   672	    actor: HealthReportActor,
   673	    healthReportId: string,
   674	  ): Promise<Record<string, unknown>> {
   675	    await this.authorize(
   676	      actor,
   677	      [
   678	        'NURSE',
   679	        'CARE_MANAGER',
   680	        'SUPERVISOR',
   681	      ],
   682	    );
   683	
   684	    const report =
   685	      await this.getReport(healthReportId);
   686	
   687	    if (
   688	      report.status !== 'DRAFT' &&
   689	      report.status !==
   690	        'REVISION_REQUIRED'
   691	    ) {
   692	      throw new BadRequestException(
   693	        'Report cannot be generated from current status',
   694	      );
   695	    }
   696	
   697	    const cutoff = new Date();
   698	
   699	    const captured =
   700	      await this.captureSnapshotData(
   701	        report.resident_id,
   702	        new Date(report.period_start),
   703	        new Date(report.period_end),
   704	        cutoff,
   705	      );
   706	
   707	    const payload = {
   708	      contractVersion:
   709	        this.snapshotContract,
   710	      sourcePolicy:
   711	        this.sourcePolicy,
   712	      residentId:
   713	        report.resident_id,
   714	      reportType:
   715	        report.report_type,
   716	      periodStart:
   717	        new Date(
   718	          report.period_start,
   719	        ).toISOString(),
   720	      periodEnd:
   721	        new Date(
   722	          report.period_end,
   723	        ).toISOString(),
   724	      sourceCutoffAt:
   725	        cutoff.toISOString(),
   726	      sources:
   727	        captured.snapshotData,
   728	    };
   729	
   730	    const serialized =
   731	      this.stableSerialize(payload);
   732	
   733	    const sourceHash =
   734	      createHash('sha256')
   735	        .update(serialized)
   736	        .digest('hex');
   737	
   738	    const snapshotId =
   739	      `health-report-snapshot-${randomUUID()}`;
   740	
   741	    return this.db.withTransaction(
   742	      async (client) => {
   743	        await client.query(
   744	          `
   745	            SELECT
   746	              pg_advisory_xact_lock(
   747	                hashtext($1)
   748	              )
   749	          `,
   750	          [
   751	            `health-report-snapshot:${healthReportId}`,
   752	          ],
   753	        );
   754	
   755	        const lockedReportResult =
   756	          await client.query<{
   757	            status: string;
   758	          }>(
   759	            `
   760	              SELECT
   761	                status
   762	              FROM health_reports
   763	              WHERE health_report_id=$1
   764	              FOR UPDATE
   765	            `,
   766	            [healthReportId],
   767	          );
   768	
   769	        const lockedReport =
   770	          lockedReportResult.rows[0];
   771	
   772	        if (!lockedReport) {
   773	          throw new BadRequestException(
   774	            'Health report not found',
   775	          );
   776	        }
   777	
   778	        if (lockedReport.status !== 'DRAFT') {
   779	          throw new BadRequestException(
   780	            'Concurrent health report generation rejected',
   781	          );
   782	        }
   783	
   784	        const snapshotVersionResult =
   785	          await client.query<{
   786	            next_version: number;
   787	          }>(
   788	            `
   789	              SELECT
   790	                COALESCE(
   791	                  MAX(snapshot_version),
   792	                  0
   793	                ) + 1 AS next_version
   794	              FROM health_report_snapshots
   795	              WHERE health_report_id=$1
   796	            `,
   797	            [healthReportId],
   798	          );
   799	
   800	        const snapshotVersion =
   801	          Number(
   802	            snapshotVersionResult
   803	              .rows[0]?.next_version ??
   804	              1,
   805	          );
   806	
   807	        const insertedSnapshot =
   808	          await client.query<HealthReportSnapshotRow>(
   809	            `
   810	              INSERT INTO health_report_snapshots (
   811	                health_report_snapshot_id,
   812	                health_report_id,
   813	                snapshot_version,
   814	                baseline_reference,
   815	                source_cutoff_at,
   816	                snapshot_data,
   817	                source_manifest,
   818	                source_hash
   819	              )
   820	              VALUES (
   821	                $1,$2,$3,$4,$5,
   822	                $6::jsonb,$7::jsonb,$8
   823	              )
   824	              RETURNING *
   825	            `,
   826	            [
   827	              snapshotId,
   828	              healthReportId,
   829	              snapshotVersion,
   830	              `report-version:${report.report_version}`,
   831	              cutoff,
   832	              serialized,
   833	              JSON.stringify(
   834	                captured.sourceManifest,
   835	              ),
   836	              sourceHash,
   837	            ],
   838	          );
   839	
   840	        await client.query(
   841	          `
   842	            UPDATE health_reports
   843	            SET
   844	              status='GENERATED',
   845	              generated_at=now(),
   846	              generated_by=$2,
   847	              generated_by_role=$3,
   848	              updated_at=now()
   849	            WHERE health_report_id=$1
   850	          `,
   851	          [
