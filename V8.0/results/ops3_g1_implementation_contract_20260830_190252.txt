     1	import { Injectable } from '@nestjs/common';
     2	import { PoolClient } from 'pg';
     3	
     4	@Injectable()
     5	export class InventoryLedgerService {
     6	  lockKey(
     7	    inventoryItemId: string,
     8	    inventoryLotId: string | null,
     9	  ): string {
    10	    return inventoryLotId
    11	      ? `INVENTORY_ITEM_LOT:${inventoryItemId}:${inventoryLotId}`
    12	      : `INVENTORY_ITEM:${inventoryItemId}`;
    13	  }
    14	
    15	  async acquireLock(
    16	    client: PoolClient,
    17	    inventoryItemId: string,
    18	    inventoryLotId: string | null,
    19	  ): Promise<void> {
    20	    await client.query(
    21	      `
    22	      SELECT pg_advisory_xact_lock(
    23	        hashtextextended($1::text, 0)
    24	      )
    25	      `,
    26	      [
    27	        this.lockKey(
    28	          inventoryItemId,
    29	          inventoryLotId,
    30	        ),
    31	      ],
    32	    );
    33	  }
    34	
    35	  async balance(
    36	    client: PoolClient,
    37	    inventoryItemId: string,
    38	    inventoryLotId: string | null,
    39	  ): Promise<number> {
    40	    const result = await client.query(
    41	      `
    42	      SELECT COALESCE(
    43	        SUM(
    44	          CASE
    45	            WHEN transaction_type IN (
    46	              'RECEIPT',
    47	              'ADJUSTMENT_IN',
    48	              'RETURN_IN'
    49	            )
    50	              THEN quantity
    51	            WHEN transaction_type IN (
    52	              'ISSUE',
    53	              'ADJUSTMENT_OUT',
    54	              'RETURN_OUT'
    55	            )
    56	              THEN -quantity
    57	            ELSE 0
    58	          END
    59	        ),
    60	        0
    61	      ) AS balance
    62	      FROM inventory_transactions
    63	      WHERE inventory_item_id = $1
    64	        AND (
    65	          $2::text IS NULL
    66	          OR inventory_lot_id = $2
    67	        )
    68	      `,
    69	      [
    70	        inventoryItemId,
    71	        inventoryLotId,
    72	      ],
    73	    );
    74	
    75	    return Number(result.rows[0]?.balance || 0);
    76	  }
    77	
    78	  signedQuantity(
    79	    transactionType: string,
    80	    quantity: number,
    81	  ): number {
    82	    if (
    83	      [
    84	        'RECEIPT',
    85	        'ADJUSTMENT_IN',
    86	        'RETURN_IN',
    87	      ].includes(transactionType)
    88	    ) {
    89	      return quantity;
    90	    }
    91	
    92	    return -quantity;
