VALIDATION_SQL_BEGIN
BEGIN READ ONLY;

SELECT
  (SELECT COUNT(*) FROM pg_namespace WHERE nspname = 'public') AS public_schema_count,

  (
    SELECT COUNT(*)
    FROM information_schema.tables
    WHERE table_schema = 'public'
  ) AS public_table_count,

  (
    SELECT COUNT(*)
    FROM information_schema.tables
    WHERE table_schema = 'public'
      AND table_name = 'residents'
  ) AS residents_table_count,

  (
    SELECT COUNT(*)
    FROM information_schema.tables
    WHERE table_schema = 'public'
      AND table_name = 'staff_actors'
  ) AS staff_actors_table_count,

  (
    SELECT COUNT(*)
    FROM information_schema.tables
    WHERE table_schema = 'public'
      AND table_name = 'resident_access_assignments'
  ) AS assignments_table_count,

  (
    SELECT COUNT(*)
    FROM information_schema.tables
    WHERE table_schema = 'public'
      AND table_name = 'resident_audit'
  ) AS resident_audit_table_count,

  (
    SELECT COUNT(*)
    FROM information_schema.tables
    WHERE table_schema = 'public'
      AND table_name = 'staff_actor_audit'
  ) AS staff_actor_audit_table_count,

  (
    SELECT
      CASE
        WHEN
          (SELECT COUNT(*) FROM information_schema.tables
           WHERE table_schema='public' AND table_name='residents') = 1
          AND
          (SELECT COUNT(*) FROM information_schema.tables
           WHERE table_schema='public' AND table_name='staff_actors') = 1
        THEN 1
        ELSE 0
      END
  ) AS core_table_pair_present;

ROLLBACK;
VALIDATION_SQL_END
