     9	export type InventoryHumanRole =
    10	  | 'CAREGIVER'
    11	  | 'NURSE'
    12	  | 'CARE_MANAGER'
    13	  | 'SUPERVISOR';
    14	
    15	export interface InventoryActor {
    16	  actorId: string;
    17	  actorRole: InventoryHumanRole;
    18	}
    19	
    20	@Injectable()
    21	export class InventoryAuthorityService {
    22	  private readonly humanRoles: InventoryHumanRole[] = [
    23	    'CAREGIVER',
    24	    'NURSE',
    25	    'CARE_MANAGER',
    26	    'SUPERVISOR',
    27	  ];
    28	
    29	  private readonly authorityRoles: InventoryHumanRole[] = [
    30	    'CARE_MANAGER',
    31	    'SUPERVISOR',
    32	  ];
    33	
    34	  constructor(private readonly database: DatabaseService) {}
    35	
    36	  async requireActor(
    37	    actorId?: string,
    38	    actorRole?: string,
    39	    client?: PoolClient,
    40	  ): Promise<InventoryActor> {
    41	    const id = String(actorId || '').trim();
    42	    const role =
    43	      String(actorRole || '').trim().toUpperCase() as InventoryHumanRole;
    44	
    45	    if (!id || !role) {
    46	      throw new UnauthorizedException('Actor context is required');
    47	    }
    48	
    49	    if (!this.humanRoles.includes(role)) {
    50	      throw new ForbiddenException('Human operational role is required');
    51	    }
    52	
    53	    const query = client
    54	      ? client.query.bind(client)
    55	      : this.database.query.bind(this.database);
    56	
    57	    const result = await query(
    58	      `
    59	      SELECT actor_id, primary_operational_role, status
    60	      FROM staff_actors
    61	      WHERE actor_id = $1
    62	      LIMIT 1
    63	      `,
    64	      [id],
    65	    );
    66	
    67	    if (!result.rowCount) {
    68	      throw new UnauthorizedException('Canonical actor not found');
    69	    }
    70	
    71	    const actor: any = result.rows[0];
    72	
    73	    if (actor.status !== 'ACTIVE') {
    74	      throw new ForbiddenException('Canonical actor is not active');
    75	    }
    76	
    77	    if (actor.primary_operational_role !== role) {
    78	      throw new ForbiddenException(
    79	        'Actor role does not match canonical role',
    80	      );
    81	    }
    82	
    83	    return {
    84	      actorId: actor.actor_id,

    87	  }

    17	interface CredentialRow {
    18	  actor_id: string;
    19	  password_hash: string;
    20	  password_salt: string;
    21	  password_iterations: number;
    22	  password_digest: string;

    29	export class AuthService {
    30	  constructor(
    31	    private readonly db:
    32	      DatabaseService,
    33	  ) {}
    34	
    35	  async login(
    36	    actorId: string,
    37	    password: string,
    38	  ) {
    39	    if (!actorId || !password) {
    40	      throw new UnauthorizedException(
    41	        'Invalid credentials',
    42	      );
    43	    }
    44	
    45	    const result =
    46	      await this.db.query<CredentialRow>(
    47	        `
    48	        SELECT
    49	          c.actor_id,
    50	          c.password_hash,
    51	          c.password_salt,
    52	          c.password_iterations,
    53	          c.password_digest,
    54	          c.locked_until,
    55	          s.primary_operational_role,
    56	          s.status
    57	        FROM auth_credentials c
    58	        JOIN staff_actors s
    59	          ON s.actor_id = c.actor_id
    60	        WHERE c.actor_id = $1
    61	        LIMIT 1
    62	        `,
    63	        [actorId],
    64	      );
    65	
    66	    const row = result.rows[0];
    67	
    68	    if (
    69	      !row ||
    70	      row.status !== 'ACTIVE' ||
    71	      (
    72	        row.locked_until &&
    73	        new Date(
    74	          row.locked_until,
    75	        ).getTime() > Date.now()
    76	      )
    77	    ) {
    78	      throw new UnauthorizedException(
    79	        'Invalid credentials',
    80	      );
    81	    }
    82	
    83	    const calculated =
    84	      pbkdf2Sync(
    85	        password,
    86	        row.password_salt,
    87	        row.password_iterations,
    88	        32,
    89	        row.password_digest,
    90	      );
    91	
    92	    const expected =
    93	      Buffer.from(
    94	        row.password_hash,
    95	        'hex',
    96	      );
    97	
    98	    const valid =
    99	      expected.length ===
   100	        calculated.length &&
   101	      timingSafeEqual(
   102	        expected,
   103	        calculated,
   104	      );
   105	
   106	    if (!valid) {
   107	      await this.db.query(
   108	        `
   109	        UPDATE auth_credentials
   110	        SET
   111	          failed_attempts =
   112	            failed_attempts + 1,
   113	          locked_until =
   114	            CASE

   159	        expires_at
   160	      )

   200	    }

   202	    const result =
   203	      await this.db.query(
   204	        `
   205	        UPDATE auth_sessions
   206	        SET
   207	          revoked_at = now(),
   208	          revoked_reason = 'LOGOUT'
   209	        WHERE session_id = $1
   210	          AND actor_id = $2
   211	          AND revoked_at IS NULL
   212	        `,
   213	        [
   214	          sessionId,
   215	          actorId,
   216	        ],
   217	      );
   218	
   219	    if (result.rowCount !== 1) {
   220	      throw new UnauthorizedException(

   226	  private issueToken(
   227	    actorId: string,
   228	    role: string,
   229	    sessionId: string,
   230	    now: number,
   231	    expiresAt: number,

   233	    const secret =
   234	      process.env.JWT_SECRET;
   235	
   236	    if (!secret || secret.length < 32) {
   237	      throw new Error(
   238	        'JWT_SECRET must contain at least 32 characters',
   239	      );
   240	    }
   241	
   242	    const encode =
   243	      (value: unknown) =>
   244	        Buffer
   245	          .from(
   246	            JSON.stringify(value),
   247	          )
   248	          .toString('base64url');
   249	
   250	    const unsigned =
   251	      `${encode({
   252	        alg: 'HS256',
   253	        typ: 'JWT',
   254	      })}.${encode({
   255	        sub: actorId,
   256	        role,
   257	        jti: sessionId,
   258	        iat: now,
   259	        exp: expiresAt,
   260	      })}`;
   261	
   262	    const signature =
   263	      createHmac(
   264	        'sha256',
   265	        secret,
   266	      )
   267	        .update(unsigned)
   268	        .digest('base64url');
   269	
   270	    return `${unsigned}.${signature}`;
   271	  }
