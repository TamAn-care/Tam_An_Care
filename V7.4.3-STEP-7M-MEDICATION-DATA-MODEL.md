TABLE 1 — medication_orders
==================================================

medication_order_id TEXT PK
resident_id TEXT FK -> residents
order_code TEXT UNIQUE NOT NULL

medication_name TEXT NOT NULL
generic_name TEXT NULL
strength TEXT NULL

dose NUMERIC NULL
dose_unit TEXT NULL
route TEXT NOT NULL
frequency TEXT NOT NULL
instructions TEXT NULL
indication TEXT NULL

prescriber_name TEXT NOT NULL
prescriber_reference TEXT NULL
prescribed_at TIMESTAMPTZ NOT NULL

effective_from TIMESTAMPTZ NULL
effective_to TIMESTAMPTZ NULL

high_risk BOOLEAN NOT NULL DEFAULT FALSE
double_check_required BOOLEAN NOT NULL DEFAULT FALSE

status TEXT NOT NULL

verified_by TEXT NULL
verified_by_role TEXT NULL
verified_at TIMESTAMPTZ NULL

created_at TIMESTAMPTZ NOT NULL DEFAULT now()
updated_at TIMESTAMPTZ NOT NULL DEFAULT now()

STATUS:
DRAFT
VERIFIED
ACTIVE
SUSPENDED
COMPLETED
CANCELLED

NO DELETE CASCADE.

==================================================

medication_administration_id TEXT PK
medication_schedule_id TEXT FK -> medication_schedules
medication_order_id TEXT FK -> medication_orders
resident_id TEXT FK -> residents

administration_code TEXT UNIQUE NOT NULL

status TEXT NOT NULL

scheduled_at TIMESTAMPTZ NOT NULL

assigned_to TEXT NULL
assigned_role TEXT NULL
assigned_at TIMESTAMPTZ NULL

accepted_at TIMESTAMPTZ NULL

ready_at TIMESTAMPTZ NULL

administered_at TIMESTAMPTZ NULL
missed_at TIMESTAMPTZ NULL
refused_at TIMESTAMPTZ NULL
held_at TIMESTAMPTZ NULL
cancelled_at TIMESTAMPTZ NULL

administration_note TEXT NULL
exception_reason TEXT NULL

created_at TIMESTAMPTZ NOT NULL DEFAULT now()
updated_at TIMESTAMPTZ NOT NULL DEFAULT now()

STATUS:
SCHEDULED
ASSIGNED
ACCEPTED
READY
ADMINISTERED
MISSED
REFUSED
HELD
CANCELLED

ASSIGNMENT ATOMICITY:

assigned_to IS NULL
assigned_role IS NULL
assigned_at IS NULL

OR

all three are NOT NULL.

NO DELETE CASCADE.

TABLE 4 — medication_double_checks
==================================================

double_check_id TEXT PK
medication_administration_id TEXT FK
medication_order_id TEXT FK
resident_id TEXT FK

checker_id TEXT NOT NULL
checker_role TEXT NOT NULL
checked_at TIMESTAMPTZ NOT NULL

result TEXT NOT NULL
check_note TEXT NULL

created_at TIMESTAMPTZ NOT NULL DEFAULT now()

RESULT:
PASSED
FAILED

UNIQUE:
medication_administration_id + checker_id

ADMINISTRATOR != CHECKER enforced in service
inside locked transaction.

AI / SYSTEM checker forbidden.
